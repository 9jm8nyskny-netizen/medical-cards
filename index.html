// 1. دالة يتم استدعاؤها عند إرسال الموظف لطلب جديد function onEmployeeSubmitRequest(e) { var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet(); var rowData = e.values; // البيانات المرسلة من النموذج // تعيين حالة الطلب إلى "قيد الانتظار" تلقائياً var lastRow = sheet.getLastRow(); sheet.getRange(lastRow, 4).setValue("Pending"); // فرضاً أن العمود الرابع هو Status // إرسال إشعار للمسؤول (أنت) بوجود طلب جديد var adminEmail = "your-email@domain.com"; // ضع بريدك الإلكتروني هنا var subject = "طلب فتح حساب جديد من: " + rowData[1]; // اسم الموظف var body = "قام الموظف " + rowData[1] + " بتقديم طلب فتح حساب.\n\n" + "البريد الإلكتروني: " + rowData[2] + "\n" + "يرجى الدخول إلى النظام للموافقة أو الرفض."; MailApp.sendEmail(adminEmail, subject, body); } // 2. دالة معالجة قرار المسؤول (موافقة / رفض) function processAdminDecision(requestId, decision, adminNotes) { var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet(); var data = sheet.getDataRange().getValues(); for (var i = 1; i < data.length; i++) { if (data[i][0] == requestId) { // البحث برقم ID var rowIndex = i + 1; var employeeEmail = data[i][2]; // بريد الموظف var employeeName = data[i][1]; // اسم الموظف if (decision === "Approved") { sheet.getRange(rowIndex, 4).setValue("Approved"); // تحديث الحالة إلى مقبول sendNotificationToEmployee(employeeEmail, employeeName, true, adminNotes); } else if (decision === "Rejected") { sheet.getRange(rowIndex, 4).setValue("Rejected"); // تحديث الحالة إلى مرفوض sendNotificationToEmployee(employeeEmail, employeeName, false, adminNotes); } break; } } } // 3. دالة إرسال النتيجة للموظف function sendNotificationToEmployee(email, name, isApproved, notes) { var subject = ""; var body = ""; if (isApproved) { subject = "تمت الموافقة على طلب فتح الحساب الخاص بك"; body = "مرحباً " + name + "،\n\nتمت الموافقة على طلب فتح الحساب بنجاح.\n" + "ملاحظات: " + (notes || "لا يوجد"); } else { subject = "تحديث بشأن طلب فتح الحساب الخاص بك"; body = "مرحباً " + name + "،\n\nنأسف لإبلاغك بأنه تم رفض طلب فتح الحساب.\n" + "سبب الرفض / ملاحظات: " + (notes || "لا يوجد"); } MailApp.sendEmail(email, subject, body); }
